\begin{frame}{خط مبنا و کنترل نشت}
\begin{itemize}
\item مرجع ثابت آموزش: احتمال منفی \pct{\NegRate} و \lr{AP=0.2034}.
\item ستاره، میانگین/تعداد امتیاز محصول و رأی انباشته از ورودی آینده حذف می‌شوند.
\item واژگان و تبدیل‌ها فقط روی آموزش برازش می‌شوند؛ آستانه با اعتبارسنجی انتخاب می‌شود.
\item مقایسه آینده: متن تنها، ساختار تنها، ترکیب، روی ردیف‌های یکسان.
\end{itemize}
\sourcenote{هیچ مدل، خوشه یا تعبیه‌ای در فاز اول آموزش داده نشده است. مرجع ثابت، عملکرد آزمون نیست.}
\end{frame}
